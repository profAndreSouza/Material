% 4. A NORMA ISO 38500
\begin{senaiframe}{Teoria • Módulo 1}{A Norma ISO/IEC 38500}
  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{tcolorbox}[senaibluecard, title={Escopo do Padrão}]
        \begin{itemize}
          \item Norma internacional focada no nível executivo/conselho.
          \item Guia diretivo para conselheiros sobre o uso aceitável da TI.
        \end{itemize}
      \end{tcolorbox}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{tcolorbox}[senaigreencard, title={Orientação}]
        \begin{itemize}

        \end{itemize}
      \end{tcolorbox}
    \end{column}
  \end{columns}
\end{senaiframe}

% 5. OS 6 PRINCÍPIOS DA ISO 38500
\begin{senaiframe}{Teoria • Módulo 1}{Os 6 Princípios da ISO/IEC 38500}
  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{tcolorbox}[senaibluecard, title={Princípios 1 a 3}]
        \begin{itemize}
          \item \textbf{1. Responsabilidade:} Clareza de deveres.
          \item \textbf{2. Estratégia:} Alinhamento do negócio.
          \item \textbf{3. Aquisição:} Razões válidas de compra.
        \end{itemize}
      \end{tcolorbox}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{tcolorbox}[senaiambercard, title={Princípios 4 a 6}]
        \begin{itemize}
          \item \textbf{4. Desempenho:} Atendimento com qualidade.
          \item \textbf{5. Conformidade:} Cumprimento de leis.
          \item \textbf{6. Comportamento Humano:} Respeito a pessoas.
        \end{itemize}
      \end{tcolorbox}
    \end{column}
  \end{columns}
\end{senaiframe}

% 6. O CICLO EDM DA ISO 38500
\begin{senaiframe}{Teoria • Módulo 1}{O Ciclo EDM da ISO 38500}
  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{tcolorbox}[senaibluecard, title={Evaluate \& Direct}]
        \begin{itemize}
          \item \textbf{Evaluate (Avaliar):} Analisar demandas do negócio.
          \item \textbf{Direct (Dirigir):} Definir políticas e estratégias.
        \end{itemize}
      \end{tcolorbox}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{tcolorbox}[senaigreencard, title={Monitor}]
        \begin{itemize}
          \item \textbf{Monitor (Monitorar):} Acompanhar os resultados práticos.
        \end{itemize}
      \end{tcolorbox}
    \end{column}
  \end{columns}
\end{senaiframe}

  \end{columns}
\end{senaiframe}

  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{tcolorbox}[senaibluecard, title={Governança \& Planejamento}]
        \begin{itemize}

          \item \textbf{APO (14 Objetivos):} Alinhar, Planejar, Organizar.

      \end{tcolorbox}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{tcolorbox}[senaiambercard, title={Execução, Serviços \& Avaliação}]

      \end{tcolorbox}
    \end{column}
  \end{columns}
\end{senaiframe}

% 9. FATORES DE DESIGN DO COBIT
\begin{senaiframe}{Teoria • Módulo 2}{Fatores de Design: Governança Sob Medida}
  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{tcolorbox}[senaibluecard, title={Tailoring (Customização)}]
        \begin{itemize}
          \item Nenhuma empresa deve aplicar os 40 objetivos de uma vez.
          \item Fatores de design (porte, setor, modelo de ameaças) selecionam o foco.
        \end{itemize}
      \end{tcolorbox}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{tcolorbox}[senaigreencard, title={Maturidade Proporcional}]
        \begin{itemize}
          \item Priorização de processos com base em dores reais.
        \end{itemize}
      \end{tcolorbox}
    \end{column}
  \end{columns}
\end{senaiframe}

% 12. HIERARQUIA DOS FRAMEWORKS
\begin{senaiframe}{Teoria • Módulo 3}{Hierarquia \& Complementaridade dos Modelos}
  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{tcolorbox}[senaibluecard, title={Níveis de Abstração}]
        \begin{itemize}
          \item \textbf{ISO 38500:} *O Quê* fazer no Conselho (Estratégico).
          \item \textbf{COBIT 2019:} *Como* governar e estruturar (Tático).

    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{tcolorbox}[senaigreencard, title={Uso Conjunto}]

      \end{tcolorbox}
    \end{column}
  \end{columns}
\end{senaiframe}

% 13. ESTUDO DE CASO PRÁTICO
\begin{senaiframe}{Estudo de Caso}{Adocão Gradual em uma PME Tecnológica}
  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{tcolorbox}[senairedcard, title={Erro Inicial}]
        \begin{itemize}
          \item Tentativa frustrada de implantar 100\% do COBIT em 6 meses.
          \item Excesso de burocracia e paralisia operacional.
        \end{itemize}
      \end{tcolorbox}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{tcolorbox}[senaigreencard, title={Solução Reajustada}]
        \begin{itemize}
          \item Seleção de 3 objetivos do COBIT e 4 práticas ITIL.
          \item Ganhos rápidos e aumento de maturidade sem trauma.
        \end{itemize}
      \end{tcolorbox}
    \end{column}
  \end{columns}
\end{senaiframe}
