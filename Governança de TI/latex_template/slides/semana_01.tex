% 3. CONCEITO DE GOVERNANÇA DE TI
\begin{senaiframe}{Teoria • Módulo 1}{O que é Governança de TI?}
  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{tcolorbox}[senaibluecard, title={Definição de Weill \& Ross (2004)}]
        \begin{itemize}
          \item Especificação dos direitos de decisão.
          \item Matriz de responsabilidade corporativa.
          \item Estímulo a comportamentos desejáveis no uso da TI.
        \end{itemize}
      \end{tcolorbox}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{tcolorbox}[senaicard, title={Definição de Fernandes \& Abreu}]
        \begin{itemize}
          \item Estrutura de relações e processos diretivos.
          \item Controle e direcionamento para atingir metas.
          \item Equilíbrio entre risco tecnológico e retorno.
        \end{itemize}
      \end{tcolorbox}
    \end{column}
  \end{columns}
\end{senaiframe}

  \end{columns}

% 13. ESTUDO DE CASO PRÁTICO
\begin{senaiframe}{Estudo de Caso}{Rede Varejista: Reestruturação do Investimento}
  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{tcolorbox}[senairedcard, title={Problema Diagnosticado}]
        \begin{itemize}
          \item Investimento de 8\% do faturamento sem retorno em vendas.
          \item Foco em trocas operacionais enquanto o e-commerce estava estagnado.
        \end{itemize}
      \end{tcolorbox}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{tcolorbox}[senaigreencard, title={Resultado com Governança}]
        \begin{itemize}
          \item Criação do Comitê de TI com CEO, CFO e CIO.
          \item Aprovação do e-commerce: +35\% de vendas totais no ano seguinte.
        \end{itemize}
      \end{tcolorbox}
    \end{column}
  \end{columns}
\end{senaiframe}
