  \begin{columns}[t]

      \begin{tcolorbox}[senaibluecard, title={Entendimento \& Diagnóstico}]
        \begin{itemize}
          \item \textbf{Fase 1 - Contexto do Negócio:} Mapear diretrizes estratégicas corporativas (metas, receita, expansão).
          \item \textbf{Fase 2 - Diagnóstico da TI:} Mapear a arquitetura atual (As-Is) e identificar limitações técnicas.
        \end{itemize}
      \end{tcolorbox}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{tcolorbox}[senaicard, title={Arquitetura Futura \& Roadmap}]
        \begin{itemize}
          \item \textbf{Fase 3 - Visão de Futuro (To-Be):} Definir sistemas, nuvem e capacidades tecnológicas necessárias.
          \item \textbf{Fase 4 - Plano de Transição:} Projetos prioritários, orçamento e cronograma de execução.
        \end{itemize}
      \end{tcolorbox}
    \end{column}
  \end{columns}

\end{senaiframe}

% 4. MATRIZ SWOT NA TI
% ------------------------------------------------------------------------------
\begin{senaiframe}{Módulo 2 • Diagnóstico}{Matriz SWOT Aplicada à TI}

  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{tcolorbox}[senaibluecard, title={Ambiente Interno}]
        \begin{itemize}
          \item \textbf{Forças (S):} Equipe técnica capacitada, infraestrutura resiliente em nuvem.
          \item \textbf{Fraquezas (W):} Falta de documentação, sistemas legados com débitos técnicos.
        \end{itemize}
      \end{tcolorbox}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{tcolorbox}[senaiambercard, title={Ambiente Externo}]
        \begin{itemize}
          \item \textbf{Oportunidades (O):} Automação por IA, SaaS, novos canais digitais de venda.
          \item \textbf{Ameaças (T):} Ataques cibernéticos (Ransomware), multas de LGPD, escassez de talentos.
        \end{itemize}
      \end{tcolorbox}
    \end{column}
  \end{columns}

\end{senaiframe}
