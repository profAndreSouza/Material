% 3. AS 4 ESTRATÉGIAS DE RESPOSTA AOS RISCOS
% ------------------------------------------------------------------------------
\begin{senaiframe}{Módulo 1 • Resposta a Riscos}{As 4 Estratégias de Tratamento}

  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{tcolorbox}[senaibluecard, title={Mitigar \& Evitar}]
        \begin{itemize}
          \item \textbf{Mitigar (Reduzir):} Implementar controles técnicos (ex: MFA, firewalls, backups).
          \item \textbf{Evitar (Eliminar):} Descontinuar sistemas legados inseguros ou processos arriscados.
        \end{itemize}
      \end{tcolorbox}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{tcolorbox}[senairedcard, title={Transferir \& Aceitar}]
        \begin{itemize}
          \item \textbf{Transferir (Compartilhar):} Seguro cibernético ou terceirização em nuvem com SLA.
          \item \textbf{Aceitar (Residual):} Assumir o risco quando o custo do controle supera o dano.
        \end{itemize}
      \end{tcolorbox}
    \end{column}
  \end{columns}

\end{senaiframe}

% 4. COMPLIANCE & SEGURANÇA DA INFORMAÇÃO
% ------------------------------------------------------------------------------
\begin{senaiframe}{Módulo 2 • Compliance}{LGPD \& ISO/IEC 27001}

  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{tcolorbox}[senaiambercard, title={LGPD (Lei 13.709/2018)}]
        \begin{itemize}
          \item Proteção de dados pessoais e direitos dos titulares.
          \item Multas administrativas de até R$ 50 milhões por infração.
          \item Governança de dados, DPO e Relatório RIPD.
        \end{itemize}
      \end{tcolorbox}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{tcolorbox}[senaicard, title={ISO/IEC 27001 (Tríade CID)}]
        \begin{itemize}
          \item \textbf{Confidencialidade:} Acesso restrito apenas a pessoas autorizadas.
          \item \textbf{Integridade:} Dados exatos, imutáveis e sem adulteração.
          \item \textbf{Disponibilidade:} Acesso garantido sempre que necessário.
        \end{itemize}
      \end{tcolorbox}
    \end{column}
  \end{columns}

\end{senaiframe}
