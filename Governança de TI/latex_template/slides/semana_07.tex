% 5. HIERARQUIA DA MEDIÇÃO
\begin{senaiframe}{Teoria • Módulo 1}{Hierarquia: Dado vs. Métrica vs. KPI}
  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{tcolorbox}[senaibluecard, title={Dado \& Métrica}]
        \begin{itemize}
          \item \textbf{Dado:} Log "Servidor caiu às 14h".
          \item \textbf{Métrica:} Tempo de parada mensal "22 minutos".
        \end{itemize}
      \end{tcolorbox}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{tcolorbox}[senaigreencard, title={KPI (Indicador Chave)}]
        \begin{itemize}
          \item \textbf{KPI:} "Disponibilidade do ERP no mês: 99,95\% (Meta: 99,90\%)".
          \item Medição com contexto e meta definida.
        \end{itemize}
      \end{tcolorbox}
    \end{column}
  \end{columns}
\end{senaiframe}

% 6. CATEGORIAS DE KPIS
\begin{senaiframe}{Teoria • Módulo 2}{As Duas Grandes Categorias de KPIs}
  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{tcolorbox}[senaibluecard, title={KPIs do Negócio (Estratégicos)}]
        \begin{itemize}
          \item Medem o impacto da TI nos resultados da empresa.
          \item Apresentados ao Conselho e Diretoria.
        \end{itemize}
      \end{tcolorbox}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{tcolorbox}[senaicard, title={KPIs Operacionais (Técnicos)}]
        \begin{itemize}
          \item Medem a eficiência e saúde da infraestrutura.
          \item Utilizados pelos gerentes e equipes de TI.
        \end{itemize}
      \end{tcolorbox}
    \end{column}
  \end{columns}
\end{senaiframe}

% 8. KPIS OPERACIONAIS
\begin{senaiframe}{Teoria • Módulo 2}{KPIs Operacionais de TI}

      \begin{tcolorbox}[senaibluecard, title={Uptime \& SLAs}]

% 9. MTTR E MTBF
\begin{senaiframe}{Teoria • Módulo 2}{Métricas Técnicas Críticas: MTTR \& MTBF}
  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{tcolorbox}[senairedcard, title={MTTR (Mean Time to Repair)}]
        \begin{itemize}
          \item Tempo médio para reparar um sistema após falha.
          \item Mede a agilidade da equipe na solução do problema.
        \end{itemize}
      \end{tcolorbox}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{tcolorbox}[senaigreencard, title={MTBF (Mean Time Between Failures)}]
        \begin{itemize}
          \item Tempo médio de operação entre duas falhas.
          \item Mede a confiabilidade e estabilidade do sistema.
        \end{itemize}
      \end{tcolorbox}
    \end{column}
  \end{columns}
\end{senaiframe}

% 10. METAS SMART E BASELINE
\begin{senaiframe}{Teoria • Módulo 3}{Linha de Base (Baseline) \& Metas SMART}
  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{tcolorbox}[senaiambercard, title={Linha de Base (Baseline)}]
        \begin{itemize}
          \item Situação atual medida antes da melhoria (ex: MTTR de 4h).
          \item Ponto de partida obrigatório para avaliar avanços.
        \end{itemize}
      \end{tcolorbox}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{tcolorbox}[senaigreencard, title={Meta SMART}]
        \begin{itemize}
          \item Específica, Mensurável, Atingível, Relevante e Temporal.
          \item *Exemplo:* Reduzir MTTR de 4h para 45min no 4º tri.
        \end{itemize}
      \end{tcolorbox}
    \end{column}
  \end{columns}
\end{senaiframe}
