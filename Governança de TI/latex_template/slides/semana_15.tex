% 3. DESAFIO DA IMPLANTAÇÃO
\begin{senaiframe}{Teoria • Módulo 1}{O Desafio da Implantação da Governança}
  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{tcolorbox}[senairedcard, title={Erro da "Big Bang"}]
        \begin{itemize}
          \item Tentar implantar todas as políticas simultaneamente.
          \item Causa paralisia operacional e rejeição cultural.
        \end{itemize}
      \end{tcolorbox}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{tcolorbox}[senaigreencard, title={Abordagem Gradual}]
        \begin{itemize}
          \item Implantação incremental por etapas (Ondas).
          \item Priorização de ganhos rápidos (*Quick Wins*).
        \end{itemize}
      \end{tcolorbox}
    \end{column}
  \end{columns}
\end{senaiframe}

% 8. FATORES CRÍTICOS DE SUCESSO (FCS)
\begin{senaiframe}{Teoria • Módulo 2}{Fatores Críticos de Sucesso (FCS)}
  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{tcolorbox}[senaibluecard, title={O que são FCS?}]

          \item Elementos indispensáveis que garantem o sucesso do projeto.
          \item Se falharem, todo o investimento em governança é perdido.

      \end{tcolorbox}
    \end{column}

      \begin{tcolorbox}[senaicard, title={Principais FCS}]
        \begin{itemize}
          \item Patrocínio executivo, foco no negócio, comunicação e quick wins.
        \end{itemize}
      \end{tcolorbox}
    \end{column}
  \end{columns}
\end{senaiframe}

% 9. PATROCÍNIO EXECUTIVO
\begin{senaiframe}{Teoria • Módulo 2}{Tone at the Top \& Foco no Negócio}
  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{tcolorbox}[senairedcard, title={Tone at the Top}]
        \begin{itemize}
          \item O CEO e os Diretores devem apoiar as regras ativamente.
          \item Se a diretoria burlar as políticas, a equipe também burlará.
        \end{itemize}
      \end{tcolorbox}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{tcolorbox}[senaigreencard, title={Foco no Negócio}]
        \begin{itemize}
          \item Governança deve simplificar os negócios, não criar burocracia inutilizável.
        \end{itemize}
      \end{tcolorbox}
    \end{column}
  \end{columns}
\end{senaiframe}

% 11. O MODELO ADKAR
\begin{senaiframe}{Teoria • Módulo 3}{O Modelo ADKAR (Prosci)}
  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{tcolorbox}[senaibluecard, title={Awareness, Desire, Knowledge}]
        \begin{itemize}
          \item \textbf{A (Awareness):} Conscientização da razão da mudança.
          \item \textbf{D (Desire):} Vontade pessoal de apoiar.
          \item \textbf{K (Knowledge):} Saber como mudar (treinamento).
        \end{itemize}
      \end{tcolorbox}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{tcolorbox}[senaigreencard, title={Ability \& Reinforcement}]
        \begin{itemize}
          \item \textbf{A (Ability):} Habilidade de praticar.
          \item \textbf{R (Reinforcement):} Sustentação e celebração.
        \end{itemize}
      \end{tcolorbox}
    \end{column}
  \end{columns}
\end{senaiframe}

% 12. ORÇAMENTO CAPEX VS OPEX
\begin{senaiframe}{Teoria • Módulo 3}{Orçamento de TI: CAPEX vs. OPEX}
  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{tcolorbox}[senaibluecard, title={CAPEX (Capital Expenditure)}]
        \begin{itemize}
          \item Compra de ativos físicos (hardware, servidores, data center).
        \end{itemize}
      \end{tcolorbox}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{tcolorbox}[senaigreencard, title={OPEX (Operational Expenditure)}]
        \begin{itemize}
          \item Custos recorrentes (assinatura de nuvem SaaS/AWS).
        \end{itemize}
      \end{tcolorbox}
    \end{column}
  \end{columns}
\end{senaiframe}
