% 6. CONCEITO DA MATRIZ RACI
\begin{senaiframe}{Teoria • Módulo 2}{A Matriz RACI}
  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{tcolorbox}[senaibluecard, title={Finalidade}]
        \begin{itemize}
          \item Atribuir responsabilidades sem dubiedade em processos.
          \item Mapear quem executa, quem aprova e quem é informado.
        \end{itemize}
      \end{tcolorbox}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{tcolorbox}[senaicard, title={Benefícios}]
        \begin{itemize}
          \item Elimina o "jogo de empurra" em incidentes graves.
          \item Dá clareza sobre o fluxo de aprovação.
        \end{itemize}
      \end{tcolorbox}
    \end{column}
  \end{columns}
\end{senaiframe}

% 7. OS 4 PAPÉIS RACI
\begin{senaiframe}{Teoria • Módulo 2}{Os 4 Papéis da Matriz RACI}
  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{tcolorbox}[senaibluecard, title={R \& A}]
        \begin{itemize}
          \item \textbf{R (Responsible):} Quem executa a tarefa prática.
          \item \textbf{A (Accountable):} O único aprovador/dono do processo.
        \end{itemize}
      \end{tcolorbox}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{tcolorbox}[senaicard, title={C \& I}]
        \begin{itemize}
          \item \textbf{C (Consulted):} Especialista ouvido previamente.
          \item \textbf{I (Informed):} Notificado após a conclusão.
        \end{itemize}
      \end{tcolorbox}
    \end{column}
  \end{columns}
\end{senaiframe}

% 8. A REGRA DE OURO DO 'A' ÚNICO
\begin{senaiframe}{Teoria • Módulo 2}{A Regra de Ouro da Matriz RACI}
  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{tcolorbox}[senairedcard, title={Exatamente UM 'A'}]
        \begin{itemize}
          \item Para cada linha de processo de TI, deve existir apenas um 'A'.
        \end{itemize}
      \end{tcolorbox}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{tcolorbox}[senaiambercard, title={Erros Comuns}]
        \begin{itemize}
          \item \textbf{Dois 'A's:} Conflito de autoridade entre gestores.
          \item \textbf{Zero 'A's:} Ninguém responde se a tarefa falhar.
        \end{itemize}
      \end{tcolorbox}
    \end{column}
  \end{columns}
\end{senaiframe}

% 9. GESTÃO DE RECURSOS (OPEX VS CAPEX)
\begin{senaiframe}{Teoria • Módulo 3}{Gestão de Recursos: OPEX vs. CAPEX}
  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{tcolorbox}[senaibluecard, title={CAPEX (Capital Expenditure)}]
        \begin{itemize}
          \item Investimento em bens de capital (compra de servidores).
          \item Entra como ativo no balanço contábil.
        \end{itemize}
      \end{tcolorbox}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{tcolorbox}[senaigreencard, title={OPEX (Operational Expenditure)}]
        \begin{itemize}
          \item Despesas operacionais recorrentes (assinatura de nuvem).
          \item Tendência de migração para modelos sob demanda.
        \end{itemize}
      \end{tcolorbox}
    \end{column}
  \end{columns}
\end{senaiframe}

% 11. PSI E USO ACEITÁVEL
\begin{senaiframe}{Teoria • Módulo 3}{PSI \& Política de Uso Aceitável}
  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{tcolorbox}[senaibluecard, title={PSI (Segurança)}]
        \begin{itemize}
          \item Regras de senhas, criptografia, MFA e acessos.
          \item Classificação da informação e gestão de incidentes.
        \end{itemize}
      \end{tcolorbox}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{tcolorbox}[senaiambercard, title={Uso Aceitável (BYOD)}]
        \begin{itemize}
          \item Diretrizes para e-mail corporativo, notebooks e redes.
          \item Regras para uso de dispositivos pessoais no trabalho.
        \end{itemize}
      \end{tcolorbox}
    \end{column}
  \end{columns}
\end{senaiframe}

% 13. ESTUDO DE CASO PRÁTICO
\begin{senaiframe}{Estudo de Caso}{Matriz RACI na Gestão de Mudanças}
  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{tcolorbox}[senaibluecard, title={Atividade}]
        \begin{itemize}
          \item Escrever código do sistema: Dev (\textbf{R}), Gestor (\textbf{A}).
          \item Aprovar deploy em Produção: Gestor (\textbf{A}), Dev (\textbf{C}).

      \end{tcolorbox}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{tcolorbox}[senaigreencard, title={Resultado}]
        \begin{itemize}

          \item Rastreabilidade total exigida pelas auditorias.

      \end{tcolorbox}
