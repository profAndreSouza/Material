% 3. CONCEITO DE SENSORIAMENTO DISCRETO
\begin{senaiframe}{Fundamentos de Campo}{O que são Sensores Discretos?}
  \begin{itemize}
    \item Dispositivos de entrada que fornecem um \textbf{sinal binário (DIGITAL: 0 ou 1, +24V DC ou 0V DC)}.
    \item Primeira camada de percepção física no chão de fábrica da célula \textbf{Smart N1}.
    \item \textbf{Funções Principais:} Detecção de presença/ausência de peças, contagem em esteiras e limites de curso de atuadores mecânicos.
  \end{itemize}

  \vspace{6pt}
  \begin{tcolorbox}[senaibluecard, title={Tipos Físicos}]
    Dividem-se entre sensores por \textbf{contato mecânico} (eletromecânicos) e sensores de \textbf{proximidade estado sólido} sem contato físico (indutivos, capacitivos, fotoelétricos e ultrassônicos).
  \end{tcolorbox}
\end{senaiframe}

% 4. SENSORES ELETROMECÂNICOS (FIM DE CURSO)
\begin{senaiframe}{Chaves Fim de Curso}{Sensores Discretos Eletromecânicos}
  Dependem do impacto físico direto entre o mecanismo em movimento e o atuador do sensor.

  \vspace{4pt}
  \begin{itemize}
    \item \textbf{Contato NA / NO (Normal Aberto):} Aberto em repouso ($0\text{V}$), fecha ao ser acionado ($24\text{V}$).
    \item \textbf{Contato NF / NC (Normal Fechado):} Fechado em repouso ($24\text{V}$), abre ao ser acionado ($0\text{V}$).
  \end{itemize}

  \vspace{6pt}
  \begin{tcolorbox}[senairedcard, title={Princípio da Falha Segura (Fail-Safe)}]
    \textbf{Uso obrigatório de contatos NF em circuitos de emergência!} Se o fio romper ou for cortado, o circuito se abre imediatamente, parando a máquina e garantindo a segurança dos operadores.
  \end{tcolorbox}
\end{senaiframe}

% 5. RICOCHETE DE CONTATO (CONTACT BOUNCE)
\begin{senaiframe}{Fenômenos Elétricos}{Ricochete de Contato (Contact Bounce) \& Debounce}
  \begin{itemize}
    \item \textbf{O Problema:} O choque mecânico dos contatos elétricos gera micro-vibrações durante $1\text{ ms}$ a $10\text{ ms}$, disparando múltiplos pulsos falsos de contagem no CLP.
    \item \textbf{Solução no CLP:} Configuração de filtro digital de entrada (\textit{Hardware Debounce Filter}) para ignorar pulsos transitórios menores que $10\text{ ms}$.
  \end{itemize}

  \vspace{6pt}
  \begin{center}
    \small
    \textbf{Vantagem do Contato Mecânico:} Imunidade total a ruídos eletromagnéticos e altas correntes.  
    \textbf{Desvantagem:} Desgaste mecânico por atrito e velocidade limitada.
  \end{center}
\end{senaiframe}

% 6. SENSORES DE PROXIMIDADE INDUTIVOS
\begin{senaiframe}{Estado Sólido}{Sensores de Proximidade Indutivos}
  Projetados para a detecção \textbf{exclusiva de materiais condutores de eletricidade (metais)} sem contato físico.

  \vspace{4pt}
  \textbf{Princípio de Operação:}
  \begin{enumerate}
    \item Um circuito oscilador LC interno gera um campo magnético de alta frequência na face sensora.
    \item A aproximação de um alvo metálico induz \textbf{Correntes de Foucault} (\textit{Eddy currents}) na superfície do metal.
    \item As correntes parasitas absorvem energia do campo, reduzindo a amplitude da oscilação.
    \item O circuito disparador (\textit{Schmitt Trigger}) detecta a queda de amplitude e comuta o transistor de saída.
  \end{enumerate}
\end{senaiframe}

% 8. FATOR DE REDUÇÃO E HISTERESE
\begin{senaiframe}{Especificações Técnicas}{Fator de Redução ($K_r$) e Histerese Operacional}
  \small
  A distância nominal ($S_n$) é especificada para \textbf{Aço Carbono Fe 360} ($K_r = 1,0$).  
  Para outros metais, a distância efetiva é $S_e = S_n \times K_r$:

  \vspace{4pt}
  \begin{center}
    \begin{tabular}{|l|c|p{4cm}|}
      \hline
      \textbf{Material do Alvo} & \textbf{Fator de Redução ($K_r$)} & \textbf{Distância Efetiva ($S_n = 10\text{ mm}$)} \\ \hline
      Aço Carbono Fe 360 & \textbf{1,00} & $10,0\text{ mm}$ \\ \hline
      Aço Inoxidável & \textbf{0,70 – 0,80} & $7,5\text{ mm}$ \\ \hline
      Latão & \textbf{0,45 – 0,50} & $4,8\text{ mm}$ \\ \hline
      Alumínio & \textbf{0,35 – 0,40} & $3,8\text{ mm}$ \\ \hline
      Cobre & \textbf{0,25 – 0,30} & $2,8\text{ mm}$ \\ \hline
    \end{tabular}
  \end{center}

  \vspace{2pt}
  \textbf{Histerese ($H$):} Diferença percentual entre o ponto de ligar e desligar do sensor. Evita oscilação da saída se a peça vibrar na borda de leitura.
\end{senaiframe}

% 9. SENSORES DE PROXIMIDADE CAPACITIVOS
\begin{senaiframe}{Estado Sólido}{Sensores de Proximidade Capacitivos}
  Detectam \textbf{qualquer material} (líquidos, plásticos, vidro, grãos, metais) com base na variação da capacitância da face sensora:

  $$C = \frac{\epsilon_0 \cdot \epsilon_r \cdot A}{d}$$

  \begin{itemize}
    \item A aproximação de qualquer corpo com constante dielétrica $\epsilon_r > 1$ altera o campo elétrico dos eletrodos concêntricos.
    \item \textbf{Constantes Dielétricas Relativas ($\epsilon_r$):}
      \begin{itemize}
        \item Água pura: $\epsilon_r = 80$ (Altíssima sensibilidade).
        \item Vidro: $\epsilon_r = 5 - 10$ | Acrílico/PET: $\epsilon_r = 3,2$.
      \end{itemize}
    \item Possuem potenciômetro traseiro para ajuste fino de sensibilidade.
  \end{itemize}
\end{senaiframe}

  Sensores eletrônicos alimentados em \textbf{Corrente Contínua (+24V DC)} seguem a convenção normatizada de cores:

  \vspace{6pt}
  \begin{center}
    \begin{tabular}{|l|c|l|}
      \hline
      \textbf{Cor do Fio} & \textbf{Sigla IEC} & \textbf{Função Elétrica} \\ \hline
      \textbf{MARROM} & BN (\textit{Brown}) & Alimentação Positiva (\textbf{+24V DC}) \\ \hline
      \textbf{AZUL} & BU (\textit{Blue}) & Alimentação Negativa (\textbf{0V / GND}) \\ \hline
      \textbf{PRETO} & BK (\textit{Black}) & Linha de Sinal / Saída Digital (\textbf{OUT}) \\ \hline
    \end{tabular}
  \end{center}

  \vspace{6pt}
  \begin{tcolorbox}[senaiambercard, title={Atenção}]
    Confundir a inversão de polaridade dos fios Marrom e Azul pode queimar o estágio de proteção interna do sensor.
  \end{tcolorbox}
\end{senaiframe}

\begin{senaiframe}{Prática & Fixação}{Diagnóstico com Multímetro \& Questões}
  \begin{block}{\textbf{Roteiro de Teste de Bancada (Sensor PNP NA)}}
    \begin{enumerate}
      \item Alimentar: Marrom = $+24\text{V}$, Azul = $0\text{V}$.
      \item Medir Tensão: Ponteira Vermelha no Fio Preto, Preta no Azul ($0\text{V}$).
      \item Sem objeto: $0\text{V}$ | Com objeto detectado: $+24\text{V}$.
    \end{enumerate}
  \end{block}

  \vspace{4pt}
  \small
  \textbf{Questão de Fixação:} Um sensor indutivo possui $S_n = 10\text{ mm}$. Para detectar uma lata de alumínio ($K_r = 0,35$), qual a distância máxima efetiva de montagem ($S_e$)?  
  \textit{Resposta:} $S_e = 10 \times 0,35 = 3,5\text{ mm}$.
\end{senaiframe}
